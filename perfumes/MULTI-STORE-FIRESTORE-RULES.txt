HELPER FUNCTIONS
----------------
function isPerfumesProgramOwner() {
  return request.auth != null
    && request.auth.uid == "Isjbto8GuLdkyh0W5VDJSfYSZF43";
}

function isStoreOwner(storeId) {
  return request.auth != null
    && exists(/databases/$(database)/documents/stores/$(storeId))
    && get(/databases/$(database)/documents/stores/$(storeId)).data.ownerUid == request.auth.uid;
}

function isStoreSubscriptionActive(storeId) {
  return exists(/databases/$(database)/documents/stores/$(storeId))
    && get(/databases/$(database)/documents/stores/$(storeId)).data.active == true
    && get(/databases/$(database)/documents/stores/$(storeId)).data.subscriptionConfigured == true
    && get(/databases/$(database)/documents/stores/$(storeId)).data.expiresAt is timestamp
    && get(/databases/$(database)/documents/stores/$(storeId)).data.expiresAt > request.time;
}

MATCH BLOCKS
------------
match /storeRequests/{uid} {
  allow create: if request.auth != null
    && request.auth.uid == uid
    && request.resource.data.ownerUid == uid
    && request.resource.data.status == "pending";

  allow read: if isPerfumesProgramOwner()
    || (request.auth != null && request.auth.uid == uid);

  allow update, delete: if isPerfumesProgramOwner();
}

match /storeUsers/{uid} {
  allow read: if isPerfumesProgramOwner()
    || (request.auth != null && request.auth.uid == uid);
  allow create, update, delete: if isPerfumesProgramOwner();
}

match /stores/{storeId} {
  // Store name/status are needed to render a public storefront and subscription gate.
  // Do not put passwords or private customer data in this document.
  allow read: if true;
  allow create, update, delete: if isPerfumesProgramOwner();

  match /products/{productId} {
    allow read: if isStoreSubscriptionActive(storeId);
    allow create, update, delete: if isStoreSubscriptionActive(storeId)
      && isStoreOwner(storeId);
  }

  match /orders/{orderId} {
    allow create: if isStoreSubscriptionActive(storeId)
      && request.auth != null
      && request.resource.data.userId == request.auth.uid;

    allow read: if isPerfumesProgramOwner()
      || (isStoreSubscriptionActive(storeId)
        && (isStoreOwner(storeId)
          || (request.auth != null && resource.data.userId == request.auth.uid)));

    allow update, delete: if isStoreSubscriptionActive(storeId)
      && (isPerfumesProgramOwner() || isStoreOwner(storeId));
  }
}

DATA MODEL
----------
- Original ZEIN & RAHAF data stays in root collections /products and /orders.
- Every additional store uses /stores/{storeId}/products and /stores/{storeId}/orders.
- A new store's storeId is the new store owner's Firebase Auth UID.
- /storeUsers/{uid} maps a store owner to their storeId.
- /storeRequests/{uid} holds a pending request until the program owner approves it.
- Approval creates /stores/{uid}, /storeUsers/{uid}, and starts a 30-day trial.
- Store link format:
  https://yoursmartsite-com.github.io/Salary-portal/perfumes/?store=STORE_ID

SECURITY NOTES
--------------
- Never create storeUsers mappings from a shop-owner client. Only the program owner may write them.
- The client-side lock is only the UI. The isStoreSubscriptionActive() rule is the actual protection for each new store's products/orders.
- Root /products and /orders retain the current single-store rules. Do not loosen those rules.
- Existing store owners may still sign in after expiry to see their management area, but writes to the store's products/orders are denied until the program owner renews the subscription.
